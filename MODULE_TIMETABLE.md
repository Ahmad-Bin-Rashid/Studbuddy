## 1. Purpose

The Timetable module allows students to view and manage their weekly class schedule. Entries are organized by day of week and time slot. Students can add, edit, and delete class entries. The module also triggers notifications before each class.

---

## 2. Data Model

```kotlin
data class TimetableEntry(
    val id: String,           // UUID — generated by AppDataStore
    val courseName: String,   // e.g. "CS101 — Algorithms"
    val dayOfWeek: Int,       // 1=Monday, 2=Tuesday, ..., 7=Sunday
    val startTime: String,    // "HH:mm" 24-hour format, e.g. "09:00"
    val endTime: String,      // "HH:mm" 24-hour format, e.g. "10:30"
    val room: String,         // optional, e.g. "Room 204A"
    val color: String         // hex string e.g. "#1565C0"
) {
    fun toJson(): JSONObject = JSONObject().apply {
        put("id", id)
        put("courseName", courseName)
        put("dayOfWeek", dayOfWeek)
        put("startTime", startTime)
        put("endTime", endTime)
        put("room", room)
        put("color", color)
    }

    companion object {
        fun fromJson(obj: JSONObject): TimetableEntry = TimetableEntry(
            id = obj.getString("id"),
            courseName = obj.getString("courseName"),
            dayOfWeek = obj.getInt("dayOfWeek"),
            startTime = obj.getString("startTime"),
            endTime = obj.getString("endTime"),
            room = obj.optString("room", ""),
            color = obj.optString("color", "#1565C0")
        )

        val DAY_NAMES = listOf("", "Monday", "Tuesday", "Wednesday",
                               "Thursday", "Friday", "Saturday", "Sunday")
    }
}
```

## 5. RecyclerView Design

- **Layout Manager:** `LinearLayoutManager(this, LinearLayoutManager.VERTICAL, false)`
- **Item Animator:** Default (`DefaultItemAnimator`)
- **Divider:** None — CardView provides visual separation
- **Sorted by:** `dayOfWeek ASC`, then `startTime ASC` (sort in Activity before passing to adapter)
- **Filter behavior:** Spinner selection triggers `loadData(filterDay)` — adapter receives pre-filtered list

## 6. Adapter Responsibilities

```kotlin
class TimetableAdapter(
    private val items: MutableList<TimetableEntry>,
    private val onEditClick: (TimetableEntry) -> Unit,
    private val onDeleteClick: (TimetableEntry) -> Unit
) : RecyclerView.Adapter<TimetableAdapter.ViewHolder>() {

    inner class ViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val viewColorStripe: View = view.findViewById(R.id.viewColorStripe)
        val textViewCourseName: TextView = view.findViewById(R.id.textViewCourseName)
        val textViewTimeSlot: TextView = view.findViewById(R.id.textViewTimeSlot)
        val textViewDayLabel: TextView = view.findViewById(R.id.textViewDayLabel)
        val textViewRoom: TextView = view.findViewById(R.id.textViewRoom)
        val imageButtonEdit: ImageButton = view.findViewById(R.id.imageButtonEditTimetable)
        val imageButtonDelete: ImageButton = view.findViewById(R.id.imageButtonDeleteTimetable)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_timetable, parent, false)
        return ViewHolder(view)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val entry = items[position]
        holder.textViewCourseName.text = entry.courseName
        holder.textViewTimeSlot.text = "${entry.startTime} – ${entry.endTime}"
        holder.textViewDayLabel.text = TimetableEntry.DAY_NAMES[entry.dayOfWeek]
        holder.textViewRoom.visibility = if (entry.room.isBlank()) View.GONE else View.VISIBLE
        holder.textViewRoom.text = entry.room
        holder.viewColorStripe.setBackgroundColor(Color.parseColor(entry.color))
        holder.imageButtonEdit.setOnClickListener { onEditClick(entry) }
        holder.imageButtonDelete.setOnClickListener { onDeleteClick(entry) }
    }

    override fun getItemCount(): Int = items.size

    fun updateList(newItems: List<TimetableEntry>) {
        items.clear()
        items.addAll(newItems)
        notifyDataSetChanged()
    }
}
```

**Adapter Rules:**
- No date formatting logic inside adapter
- No data access inside adapter
- Color parsing (`Color.parseColor`) is the only non-binding logic permitted

## 8. Dialog Designs

### Add/Edit Dialog

```kotlin
private fun showAddDialog() {
    val dialogView = LayoutInflater.from(this).inflate(R.layout.dialog_add_timetable, null)

    val editTextCourseName = dialogView.findViewById<EditText>(R.id.editTextCourseName)
    val spinnerDay = dialogView.findViewById<Spinner>(R.id.spinnerDay)
    val spinnerStartTime = dialogView.findViewById<Spinner>(R.id.spinnerStartTime)
    val spinnerEndTime = dialogView.findViewById<Spinner>(R.id.spinnerEndTime)
    val editTextRoom = dialogView.findViewById<EditText>(R.id.editTextRoom)

    // Populate Day Spinner: "Monday" ... "Sunday"
    val dayAdapter = ArrayAdapter(this, android.R.layout.simple_spinner_item,
        listOf("Monday","Tuesday","Wednesday","Thursday","Friday","Saturday","Sunday"))
    spinnerDay.adapter = dayAdapter

    // Populate Time Spinners: "08:00", "08:30", ..., "22:00"
    val timeSlots = generateTimeSlots(startHour = 8, endHour = 22, stepMinutes = 30)
    val timeAdapter = ArrayAdapter(this, android.R.layout.simple_spinner_item, timeSlots)
    spinnerStartTime.adapter = timeAdapter
    spinnerEndTime.adapter = timeAdapter
    spinnerEndTime.setSelection(1) // default end = startTime + 30min

    AlertDialog.Builder(this)
        .setTitle("Add Class")
        .setView(dialogView)
        .setPositiveButton("Save") { dialog, _ ->
            val courseName = editTextCourseName.text.toString().trim()
            val day = spinnerDay.selectedItemPosition + 1
            val startTime = spinnerStartTime.selectedItem.toString()
            val endTime = spinnerEndTime.selectedItem.toString()
            val room = editTextRoom.text.toString().trim()

            if (!validateTimetableInput(courseName, startTime, endTime)) return@setPositiveButton

            val entry = TimetableEntry(
                id = UUID.randomUUID().toString(),
                courseName = courseName,
                dayOfWeek = day,
                startTime = startTime,
                endTime = endTime,
                room = room,
                color = "#1565C0"
            )
            saveEntry(entry)
        }
        .setNegativeButton("Cancel", null)
        .show()
}
```

## 9. Business Logic

```kotlin
// Generate time slots for Spinner
private fun generateTimeSlots(startHour: Int, endHour: Int, stepMinutes: Int): List<String> {
    val slots = mutableListOf<String>()
    var h = startHour
    var m = 0
    while (h < endHour || (h == endHour && m == 0)) {
        slots.add(String.format("%02d:%02d", h, m))
        m += stepMinutes
        if (m >= 60) { m = 0; h++ }
    }
    return slots
}

// Validate: end time must be after start time
private fun validateTimetableInput(courseName: String, startTime: String, endTime: String): Boolean {
    if (courseName.isBlank()) {
        Toast.makeText(this, "Course name cannot be empty", Toast.LENGTH_SHORT).show()
        return false
    }
    if (startTime >= endTime) {
        Toast.makeText(this, "End time must be after start time", Toast.LENGTH_SHORT).show()
        return false
    }
    return true
}

// Save entry
private fun saveEntry(entry: TimetableEntry) {
    AppDataStore.addTimetableEntry(entry)
    NotificationScheduler.scheduleTimetableNotification(this, entry)
    loadData(currentFilterDay)
}

// Delete entry
private fun deleteEntry(entryId: String) {
    NotificationScheduler.cancelTimetableNotification(this, entryId)
    AppDataStore.deleteTimetableEntry(entryId)
    loadData(currentFilterDay)
}

// Load data with optional day filter
private fun loadData(filterDay: Int = 0) {
    val all = AppDataStore.getTimetableList()
    val filtered = if (filterDay == 0) all else all.filter { it.dayOfWeek == filterDay }
    val sorted = filtered.sortedWith(compareBy({ it.dayOfWeek }, { it.startTime }))
    timetableList.clear()
    timetableList.addAll(sorted)
    timetableAdapter.updateList(timetableList)
    updateEmptyState()
}
```

## 10. Edge Cases

| Case | Handling |
|---|---|
| Course name is blank | Show Toast "Course name cannot be empty", keep dialog open |
| End time equals start time | Show Toast "End time must be after start time" |
| End time before start time | Show Toast "End time must be after start time" |
| Duplicate class at same day/time | Allowed — user may have back-to-back classes |
| Course name > 100 chars | EditText `maxLength="100"` set in XML |
| No entries (empty state) | Show `textViewEmptyTimetable`, hide `recyclerViewTimetable` |
| Room field blank | `textViewRoom` is `GONE` in item layout |
| Invalid color hex | Default to `#1565C0` — color selection is via predefined Spinner |

## 11. Data Flow

```
onResume()
    → AppDataStore.getTimetableList()
    → sort by dayOfWeek, startTime
    → timetableAdapter.updateList(sorted)
    → updateEmptyState()

Add:
    → dialog → validate → TimetableEntry(id=UUID...)
    → AppDataStore.addTimetableEntry(entry)
    → NotificationScheduler.scheduleTimetableNotification(entry)
    → loadData()

Edit:
    → dialog pre-filled → validate → entry.copy(...)
    → AppDataStore.updateTimetableEntry(entry)
    → NotificationScheduler.rescheduleTimetableNotification(entry)
    → loadData()

Delete:
    → confirm → AppDataStore.deleteTimetableEntry(id)
    → NotificationScheduler.cancelTimetableNotification(id)
    → loadData()
```

## 13. UI States

| State | UI |
|---|---|
| Loading (onResume start) | No loading spinner — data is synchronous |
| Empty list | `textViewEmptyTimetable` VISIBLE, `recyclerViewTimetable` GONE |
| Non-empty list | `textViewEmptyTimetable` GONE, `recyclerViewTimetable` VISIBLE |
| Filter = specific day, no matches | Same as empty list |

```kotlin
private fun updateEmptyState() {
    if (timetableList.isEmpty()) {
        textViewEmptyTimetable.visibility = View.VISIBLE
        recyclerViewTimetable.visibility = View.GONE
    } else {
        textViewEmptyTimetable.visibility = View.GONE
        recyclerViewTimetable.visibility = View.VISIBLE
    }
}
```
